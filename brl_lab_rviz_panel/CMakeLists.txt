cmake_minimum_required(VERSION 3.0.2)
project(brl_lab_rviz_panel)
find_package(catkin REQUIRED COMPONENTS roscpp rviz pluginlib actionlib brl_interfaces rci_action_manager geometry_msgs moveit_msgs sensor_msgs trajectory_msgs std_srvs std_msgs)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
add_definitions(-DQT_NO_KEYWORDS)
catkin_package()
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(${PROJECT_NAME} src/brl_action_panel.cpp include/brl_lab_rviz_panel/brl_action_panel.hpp)
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} Qt5::Widgets)
add_dependencies(${PROJECT_NAME} ${catkin_EXPORTED_TARGETS})
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES plugin_description.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
