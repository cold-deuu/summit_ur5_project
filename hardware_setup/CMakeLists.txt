cmake_minimum_required(VERSION 3.0.2)
project(summit_ur5_hardware_setup)
find_package(catkin REQUIRED)
catkin_package()
file(GLOB HARDWARE_PYTHON_SCRIPTS "${CMAKE_CURRENT_SOURCE_DIR}/*.py")
catkin_install_python(PROGRAMS ${HARDWARE_PYTHON_SCRIPTS}
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
file(GLOB HARDWARE_LAUNCH_FILES "${CMAKE_CURRENT_SOURCE_DIR}/*.launch")
install(FILES ${HARDWARE_LAUNCH_FILES} README.md FT_GRAVITY.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY calibration field DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
